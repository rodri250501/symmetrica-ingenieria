/* ================================================================
   NAVEGACIÓN
   ================================================================ */
function switchTab(tab){
    document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    document.getElementById('view-' + tab).classList.add('active');
    const t = document.getElementById('tab-' + tab);
    if (t) t.classList.add('active');
}

/* ================================================================
   BARRA DE ESTADO
   ================================================================ */
function actualizarBarraEstado(calculado){
    const dot = document.getElementById('statusDot');
    const texto = document.getElementById('statusTexto');
    if (calculado){
        dot.classList.add('ok'); dot.classList.remove('pending');
        texto.textContent = 'Calculado ✓';
        document.getElementById('statusHora').textContent = new Date().toLocaleTimeString('es-BO', {hour:'2-digit', minute:'2-digit'});
        document.getElementById('tabsInfo').textContent = 'Calculado';
    } else {
        dot.classList.add('pending'); dot.classList.remove('ok');
        texto.textContent = 'Sin calcular';
        document.getElementById('statusHora').textContent = '—';
        document.getElementById('tabsInfo').textContent = 'Sin calcular';
    }
}

/* ================================================================
   GRUPOS DE BARRAS
   ================================================================ */
let contadorGrupos = 0;

function obtenerOpcionesDiametros(selectedValue){
    return diametros.map(d =>
        `<option value="${d.value}" ${d.value === selectedValue ? 'selected' : ''}>${d.label}</option>`
    ).join('');
}

function agregarGrupo(diametro, nBarras){
    contadorGrupos++;
    const id = contadorGrupos;
    const container = document.getElementById('gruposContainer');
    if (!container) return;

    const div = document.createElement('div');
    div.className = 'grupo-card';
    div.id = `grupo-${id}`;

    const diametroVal = diametro !== undefined && diametro !== null ? diametro : 12;
    const nBarrasVal = nBarras !== undefined && nBarras !== null ? nBarras : 2;

    div.innerHTML = `
        <span class="num">Grupo ${id}</span>
        <div class="field">
            <label>N° de barras</label>
            <input type="number" id="nBarras_${id}" value="${nBarrasVal}" min="0" step="1" oninput="calcular()" />
        </div>
        <div class="field">
            <label>Diámetro (mm)</label>
            <select id="diametro_${id}" onchange="calcular()">
                ${obtenerOpcionesDiametros(diametroVal)}
            </select>
        </div>
        <button class="btn sm danger" onclick="eliminarGrupo(${id})" title="Eliminar grupo">
            <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" style="width:12px;height:12px;"><path d="M18 6 6 18"/><path d="m6 6 12 12"/></svg>
            Quitar
        </button>
    `;

    container.appendChild(div);
    actualizarContadorGrupos();
    calcular();
}

function eliminarGrupo(id){
    const el = document.getElementById(`grupo-${id}`);
    if (el) el.remove();
    actualizarContadorGrupos();
    calcular();
}

function limpiarGrupos(){
    const container = document.getElementById('gruposContainer');
    container.innerHTML = '';
    contadorGrupos = 0;
    agregarGrupo(12, 2);
}

function actualizarContadorGrupos(){
    const n = document.querySelectorAll('#gruposContainer .grupo-card').length;
    document.getElementById('statusGrupos').textContent = `Grupos: ${n}`;
}

/* ================================================================
   CÁLCULO PRINCIPAL
   ================================================================ */
function calcular(){
    const asReq = parseFloat(document.getElementById('asReq').value);
    if (isNaN(asReq) || asReq < 0) {
        alert('Complete el área de acero requerida.');
        return;
    }

    const grupos = [];
    document.querySelectorAll('#gruposContainer .grupo-card').forEach(el => {
        const id = el.id.split('-')[1];
        const nBarras = parseInt(document.getElementById(`nBarras_${id}`).value) || 0;
        const diametro = parseFloat(document.getElementById(`diametro_${id}`).value) || 0;
        if (nBarras > 0 && diametro > 0) {
            const area = (Math.PI / 4) * Math.pow(diametro / 10, 2);
            grupos.push({ id, nBarras, diametro, areaUnitaria: area, areaTotal: nBarras * area });
        }
    });

    const areaTotal = grupos.reduce((sum, g) => sum + g.areaTotal, 0);
    const diferencia = areaTotal - asReq;
    const cumple = diferencia >= -0.001;

    document.getElementById('statusResumen').textContent = `As req: ${asReq.toFixed(2)} cm² · As prov: ${areaTotal.toFixed(2)} cm²`;
    actualizarBarraEstado(true);
    actualizarContadorGrupos();

    const container = document.getElementById('resultadosContainer');

    let html = `
        <div class="kpi">
            <div class="kpi-box"><div class="k">Área requerida</div><div class="v">${asReq.toFixed(3)} cm²</div></div>
            <div class="kpi-box"><div class="k">Área calculada</div><div class="v">${areaTotal.toFixed(3)} cm²</div></div>
            <div class="kpi-box ${cumple ? 'good' : 'bad'}"><div class="k">Estado</div><div class="v">${cumple ? 'CUMPLE' : 'NO CUMPLE'}</div></div>
        </div>`;

    if (grupos.length > 0) {
        html += `
        <table class="tbl">
            <thead><tr><th>Grupo</th><th>N° barras</th><th>Diámetro (mm)</th><th>Área unitaria (cm²)</th><th>Área total (cm²)</th></tr></thead>
            <tbody>`;
        grupos.forEach(g => {
            html += `<tr>
                <td class="left">Grupo ${g.id}</td>
                <td>${g.nBarras}</td>
                <td>${g.diametro.toFixed(2)}</td>
                <td>${g.areaUnitaria.toFixed(4)}</td>
                <td>${g.areaTotal.toFixed(3)}</td>
            </tr>`;
        });
        html += `
            <tr class="total">
                <td class="left" colspan="4">Área total provista</td>
                <td>${areaTotal.toFixed(3)} cm²</td>
            </tr>
            </tbody></table>`;

        const detalle = !cumple
            ? `<span class="badge fail">Faltan ${Math.abs(diferencia).toFixed(3)} cm²</span>`
            : (diferencia > 0
                ? `<span class="badge warn">Exceso de ${diferencia.toFixed(3)} cm²</span>`
                : `<span class="badge ok">Cumple exactamente</span>`);
        html += `<div style="display:flex;align-items:center;gap:8px;margin-top:8px;">
            <span class="mono" style="font-size:11px;color:var(--ink-soft);">As prov − As req = ${diferencia.toFixed(3)} cm²</span>
            ${detalle}
        </div>`;
    } else {
        html += `<div class="empty">
            <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5"><path d="M4 19V10"/><path d="M11 19V5"/><path d="M18 19v-7"/><line x1="3" y1="19" x2="21" y2="19"/></svg>
            <div style="font-size:12px;">No hay grupos de barras definidos.</div>
        </div>`;
    }

    // Tabla de áreas de referencia (solo diámetros en uso)
    const diametrosUsados = new Set(grupos.map(g => g.diametro));
    const diametrosMostrar = diametros.filter(d => diametrosUsados.has(d.value));
    if (diametrosMostrar.length === 0) {
        diametrosMostrar.push(...diametros.filter(d => [10, 12, 16, 20, 25].includes(d.value)));
    }
    html += `
        <div style="margin-top:12px;padding-top:10px;border-top:1px solid var(--line-soft);">
            <h4 style="font-size:11px;font-weight:600;margin:0 0 6px;color:var(--ink);">Áreas de barras individuales</h4>
            <table class="tbl">
                <thead><tr><th>Diámetro (mm)</th><th>Área (cm²)</th></tr></thead>
                <tbody>`;
    diametrosMostrar.forEach(d => {
        const area = (Math.PI / 4) * Math.pow(d.value / 10, 2);
        html += `<tr><td class="left">${d.label}</td><td>${area.toFixed(4)}</td></tr>`;
    });
    html += `</tbody></table>
        </div>`;

    container.innerHTML = html;
}
