/* ================================================================
   CÁLCULO DE ACERO
   ================================================================ */
function calcularAcero(Mu_kN, b, h, fck, fy, rec = 5) {
    const d = h - rec;
    if (d <= 0) return { As: 0, rho: 0, a: 0, d, valida: false, mensaje: 'd<=0' };
    const fck_mpa = parseFloat(fck);
    const fy_mpa = parseFloat(fy);
    const fck_kNcm2 = fck_mpa / 10;
    const fy_kNcm2 = fy_mpa / 10;
    const Mu_kNcm = Mu_kN * 100;
    const phi = 0.9;
    const beta1 = (fck_mpa <= 28) ? 0.85 : Math.max(0.65, 0.85 - 0.05 * (fck_mpa - 28) / 7);
    const A = (phi * fy_kNcm2 * fy_kNcm2) / (2 * 0.85 * fck_kNcm2 * b);
    const B = -(phi * fy_kNcm2 * d);
    const C = Mu_kNcm;
    const disc = B * B - 4 * A * C;
    if (disc < 0) return { As: 0, rho: 0, a: 0, d, valida: false, mensaje: 'disc<0' };
    const As1 = (-B + Math.sqrt(disc)) / (2 * A);
    const As2 = (-B - Math.sqrt(disc)) / (2 * A);
    let As = (As1 > 0 && As1 < As2) ? As1 : As2;
    if (As <= 0) As = (As1 > 0) ? As1 : As2;
    const a = (As * fy_kNcm2) / (0.85 * fck_kNcm2 * b);
    if (a > 0.85 * beta1 * d) return { As, rho: As / (b * d), a, d, valida: false, mensaje: 'sobrearmado' };
    const rho = As / (b * d);
    const rho_min = Math.max(0.25 * Math.sqrt(fck_mpa) / fy_mpa, 1.4 / fy_mpa);
    const rho_max = 0.75 * (0.85 * fck_mpa * beta1 * 0.003) / (fy_mpa * (0.003 + 0.005));
    let valida = true, mensaje = 'OK';
    if (rho < rho_min) { valida = false; mensaje = `rho<rho_min`; }
    else if (rho > rho_max) { valida = false; mensaje = `rho>rho_max`; }
    return { As, rho, a, d, valida, mensaje, rho_min, rho_max };
}

/* ================================================================
   ÁREAS DE BARRAS (φ6, φ8, φ10, φ12, φ16, φ20, φ25)
   ================================================================ */
const areaBarras = {
    6: 0.283,
    8: 0.503,
    10: 0.785,
    12: 1.131,
    16: 2.011,
    20: 3.142,
    25: 4.909
};

function generarCombinaciones(AsRequerido, diametros, limite = 10) {
    if (!Number.isFinite(AsRequerido) || AsRequerido <= 0 ||
        !Array.isArray(diametros) || diametros.length === 0 ||
        diametros.some(d => !Object.hasOwn(areaBarras, d)) ||
        !Number.isInteger(limite) || limite <= 0) return [];
    const diametrosUnicos = [...new Set(diametros.map(Number))].sort((a, b) => a - b);
    const areas = diametrosUnicos.map(d => Math.round(areaBarras[d] * 1000));
    const requerido = Math.ceil(AsRequerido * 1000);
    if (!Number.isSafeInteger(requerido)) return [];
    const maxArea = Math.max(Math.ceil(requerido * 1.3), requerido + Math.max(...areas));
    const combinaciones = [];
    const cantidades = new Array(areas.length).fill(0);
    let visitados = 0;
    const maxVisitados = 25000;
    function buscar(areaAcumulada, indiceActual) {
        if (++visitados > maxVisitados || combinaciones.length >= limite) return;
        if (areaAcumulada >= requerido) {
            const detalle = diametrosUnicos.flatMap((diametro, i) => cantidades[i] > 0
                ? [{ diametro, cantidad: cantidades[i], area: cantidades[i] * areas[i] / 1000 }]
                : []);
            const areaTotal = areaAcumulada / 1000;
            combinaciones.push({
                areaTotal,
                totalBarras: detalle.reduce((t, it) => t + it.cantidad, 0),
                excesoPorc: (areaTotal - AsRequerido) / AsRequerido * 100,
                combinacion: detalle
            });
            return;
        }
        if (indiceActual >= areas.length) return;
        const maxBarras = Math.floor((maxArea - areaAcumulada) / areas[indiceActual]);
        for (let n = 0; n <= maxBarras && visitados < maxVisitados && combinaciones.length < limite; n++) {
            cantidades[indiceActual] = n;
            buscar(areaAcumulada + n * areas[indiceActual], indiceActual + 1);
        }
        cantidades[indiceActual] = 0;
    }
    buscar(0, 0);
    return combinaciones.sort((a, b) => a.areaTotal - b.areaTotal || a.totalBarras - b.totalBarras);
}

/* ================================================================
   OPTIMIZACIÓN CON PARETO
   ================================================================ */
function optimizar() {
    const Mu_kN = parseFloat(document.getElementById('Mu').value);
    const L = parseFloat(document.getElementById('L').value);
    const fck = document.getElementById('fck').value;
    const fy = document.getElementById('fy').value;
    const costoHormigon = parseFloat(document.getElementById('costH').value);
    const costoAcero = parseFloat(document.getElementById('costA').value);
    const rec = parseFloat(document.getElementById('rec').value);
    const bMin = parseFloat(document.getElementById('bMin').value);
    const bMax = parseFloat(document.getElementById('bMax').value);
    const hMin = parseFloat(document.getElementById('hMin').value);
    const hMax = parseFloat(document.getElementById('hMax').value);
    const checkboxes = document.querySelectorAll('.checkbox-item input[type="checkbox"]:checked');
    const diametros = Array.from(checkboxes).map(cb => parseInt(cb.value)).sort((a, b) => a - b);

    const valoresPositivos = [Mu_kN, L, Number(fck), Number(fy), bMin, bMax, hMin, hMax, rec];
    let error = '';
    if (valoresPositivos.some(v => !Number.isFinite(v) || v <= 0) ||
        [costoHormigon, costoAcero].some(v => !Number.isFinite(v) || v < 0)) {
        error = 'Completa los parámetros con números válidos.';
    } else if (bMin > bMax || hMin > hMax) {
        error = 'El mínimo de cada rango debe ser ≤ al máximo.';
    } else if (rec >= hMin) {
        error = 'El recubrimiento debe ser menor que la altura mínima.';
    } else if ((Math.floor(bMax - bMin) + 1) * (Math.floor((hMax - hMin) / 2) + 1) > 5000) {
        error = 'Reduce los rangos (máx. 5000 secciones).';
    } else if (diametros.length === 0) {
        error = 'Selecciona al menos un diámetro de barra.';
    }
    if (error) return { puntosValidos: [], mejor: null, error };

    let candidatos = [];
    let motivos = { dNegativo: 0, discNegativo: 0, sobrearmado: 0, rhoMin: 0, rhoMax: 0 };
    let peorRho = null;

    for (let b = bMin; b <= bMax; b += 1) {
        for (let h = hMin; h <= hMax; h += 2) {
            const r = calcularAcero(Mu_kN, b, h, fck, fy, rec);
            if (!r.valida) {
                if (r.mensaje === 'd<=0') motivos.dNegativo++;
                else if (r.mensaje === 'disc<0') motivos.discNegativo++;
                else if (r.mensaje === 'sobrearmado') motivos.sobrearmado++;
                else if (r.mensaje.startsWith('rho<rho_min')) {
                    motivos.rhoMin++;
                    if (!peorRho || r.rho > peorRho.rho) peorRho = { b, h, rho: r.rho, rho_min: r.rho_min, As: r.As };
                } else if (r.mensaje.startsWith('rho>rho_max')) motivos.rhoMax++;
                continue;
            }
            const { As, rho, a, d } = r;
            const Vol = (b * h / 10000) * L;
            const pesoAcero = (As / 10000) * L * 7850;
            const costoConcreto = Vol * costoHormigon;
            const costoAceroMat = pesoAcero * costoAcero;
            const costoTotal = costoConcreto + costoAceroMat;
            const robustez = Math.max(0, Math.min(1, (r.rho_max - rho) / (r.rho_max - r.rho_min)));
            candidatos.push({
                b, h, d, As, rho, a,
                costoConcreto, costoAceroMat, costo: costoTotal,
                peso: pesoAcero, robustez, Vol, valida: true,
                rho_min: r.rho_min, rho_max: r.rho_max
            });
        }
    }

    if (candidatos.length === 0) {
        mostrarDiagnostico(motivos, peorRho, { bMin, bMax, hMin, hMax, Mu_kN });
        return { puntosValidos: [], mejor: null };
    }

    let paretoOptimos = [];
    for (let c1 of candidatos) {
        let dominado = candidatos.some(c2 =>
            c2.costo <= c1.costo && c2.robustez >= c1.robustez &&
            (c2.costo < c1.costo || c2.robustez > c1.robustez));
        if (!dominado) paretoOptimos.push(c1);
    }
    const optima_costo = candidatos.reduce((a, b) => a.costo < b.costo ? a : b);
    const optima_economica = paretoOptimos.reduce((best, curr) => {
        const sc = curr.costo * (1 - 0.25 * curr.robustez);
        const sb = best.costo * (1 - 0.25 * best.robustez);
        return sc < sb ? curr : best;
    });
    const optima_robusta = paretoOptimos.reduce((a, b) =>
        a.robustez > b.robustez ? a : (a.robustez === b.robustez && a.costo < b.costo ? a : b));

    let mejor;
    if (criterioActual === 'optimal') mejor = optima_costo;
    else if (criterioActual === 'economic') mejor = optima_economica;
    else mejor = optima_robusta;

    const top10 = [...candidatos].sort((a, b) => a.costo - b.costo).slice(0, 10);

    return {
        puntosValidos: candidatos, mejor,
        optima_costo, optima_economica, optima_robusta,
        top10, paretoOptimos,
        params: { Mu_kN, L, fck, fy, costoHormigon, costoAcero, rec,
                  bMin, bMax, hMin, hMax, diametros }
    };
}

/* ================================================================
   DIAGNÓSTICO
   ================================================================ */
function mostrarDiagnostico(motivos, peorRho, params) {
    const total = motivos.dNegativo + motivos.discNegativo + motivos.sobrearmado + motivos.rhoMin + motivos.rhoMax;
    let causaPrincipal = Object.entries(motivos).sort((a, b) => b[1] - a[1])[0];
    let mensaje = '', sugerencia = '';
    if (causaPrincipal[0] === 'rhoMin' && motivos.rhoMin === total) {
        mensaje = `Con Mu = ${params.Mu_kN} kN·m, todas las secciones están <strong>sobredimensionadas</strong>.`;
        sugerencia = peorRho ? `Prueba reduciendo h_max a ~${Math.max(20, Math.round(peorRho.h * Math.sqrt(peorRho.rho_min / peorRho.rho)))} cm.` : `Reduce h_max a 20-40 cm.`;
    } else if (causaPrincipal[0] === 'rhoMax' && motivos.rhoMax === total) {
        mensaje = `Todas las secciones están <strong>subdimensionadas</strong> (cuantía máxima superada).`;
        sugerencia = `Aumenta b_max y h_max, o sube f'c.`;
    } else if (causaPrincipal[0] === 'sobrearmado' && motivos.sobrearmado === total) {
        mensaje = `Todas las secciones resultan <strong>sobrearmadas</strong>.`;
        sugerencia = `Aumenta el ancho b o el peralte h.`;
    } else if (causaPrincipal[0] === 'dNegativo' && motivos.dNegativo === total) {
        mensaje = `El peralte efectivo d = h − rec es muy pequeño.`;
        sugerencia = `Aumenta h_min.`;
    } else {
        mensaje = `No se encontraron secciones válidas.`;
        sugerencia = `Amplía el rango de b y h.`;
    }
    const html = `
        <div class="diag-box">
            <h3>⚠️ No se encontraron secciones válidas</h3>
            <p style="margin-bottom: 10px;">${mensaje}</p>
            <div class="diag-sugerencia"><strong>💡 Sugerencia:</strong> ${sugerencia}</div>
            <div class="diag-detail">${total} combinaciones — ${motivos.rhoMin} bajo cuantía mín., ${motivos.rhoMax} sobre cuantía máx., ${motivos.sobrearmado} sobrearmadas, ${motivos.dNegativo} peralte inválido, ${motivos.discNegativo} sección insuficiente.</div>
        </div>`;
    document.getElementById('resultadosContainer').innerHTML = html;
    document.getElementById('resultadosContainer').style.display = 'block';
    document.getElementById('resultadosVacio').style.display = 'none';
    document.getElementById('chartsSection').style.display = 'none';
    document.getElementById('chartsVacio').style.display = 'block';
    document.getElementById('statusSeccion').textContent = 'b×h: — × — cm';
    setStatus('Sin secciones válidas — ver diagnóstico', 'fail');
    document.getElementById('tabsInfo').textContent = 'Sin secciones';
    switchTab('resultados');
}

function renderCharts(puntos, mejor, paretoOptimos, resultado) {
    const ctxScatter = document.getElementById('costChart').getContext('2d');
    if (chartScatter) chartScatter.destroy();

    const bValues = puntos.map(p => p.b);
    const minB = Math.min(...bValues);
    const maxB = Math.max(...bValues);
    const rangeB = (maxB - minB) || 1;

    const scatterData = puntos.map(p => ({ x: p.h, y: p.costo, b: p.b }));
    const colores = scatterData.map(p => {
        const ratio = (p.b - minB) / rangeB;
        const r = Math.round(50 + 205 * ratio);
        const bl = Math.round(255 - 205 * ratio);
        return `rgba(${r}, 80, ${bl}, 0.5)`;
    });

    const datasets = [{
        label: 'Secciones válidas',
        data: scatterData,
        backgroundColor: colores,
        borderColor: colores.map(c => c.replace('0.5', '1')),
        pointRadius: 4, pointHoverRadius: 7,
    }];

    if (paretoOptimos && paretoOptimos.length > 0) {
        datasets.push({
            label: '◆ Frontera Pareto',
            data: paretoOptimos.map(p => ({ x: p.h, y: p.costo })),
            backgroundColor: 'rgba(14, 108, 179, 0.3)',
            borderColor: 'rgba(14, 108, 179, 1)',
            pointRadius: 6, pointBorderWidth: 2, pointBorderColor: '#fff',
            showLine: false,
        });
    }
    if (resultado && resultado.optima_costo) {
        datasets.push({ label: '🎯 Óptima (costo)', data: [{ x: resultado.optima_costo.h, y: resultado.optima_costo.costo }], backgroundColor: 'rgba(18, 128, 63, 0.9)', borderColor: 'rgba(18, 128, 63, 1)', pointRadius: 10, pointHoverRadius: 14, pointBorderColor: 'white', pointBorderWidth: 2, pointStyle: 'triangle' });
        datasets.push({ label: '💰 Económica + Robusta', data: [{ x: resultado.optima_economica.h, y: resultado.optima_economica.costo }], backgroundColor: 'rgba(151, 96, 10, 0.9)', borderColor: 'rgba(151, 96, 10, 1)', pointRadius: 10, pointHoverRadius: 14, pointBorderColor: 'white', pointBorderWidth: 2, pointStyle: 'rectRot' });
        datasets.push({ label: '🏗️ Máxima Robustez', data: [{ x: resultado.optima_robusta.h, y: resultado.optima_robusta.costo }], backgroundColor: 'rgba(14, 108, 179, 0.9)', borderColor: 'rgba(14, 108, 179, 1)', pointRadius: 10, pointHoverRadius: 14, pointBorderColor: 'white', pointBorderWidth: 2, pointStyle: 'rect' });
    }
    if (mejor) {
        datasets.push({ label: '★ Seleccionada', data: [{ x: mejor.h, y: mejor.costo }], backgroundColor: 'rgba(0,0,0,0)', borderColor: 'rgba(191, 42, 42, 1)', pointRadius: 18, pointHoverRadius: 20, pointBorderWidth: 3 });
    }

    chartScatter = new Chart(ctxScatter, {
        type: 'scatter',
        data: { datasets },
        plugins: [whiteBgPlugin],
        options: {
            responsive: true, maintainAspectRatio: false,
            plugins: {
                legend: { display: true, labels: { usePointStyle: true, padding: 12, font: { size: 10 } } },
                tooltip: { callbacks: { label: ctx => { const r = ctx.raw; if (r.b !== undefined) return `b=${r.b}cm, h=${r.x}cm, $${r.y.toFixed(0)}`; return `h=${r.x}cm, $${r.y.toFixed(0)}`; } } }
            },
            scales: {
                x: { title: { display: true, text: 'Peralte (h) [cm]', font: { weight: 'bold', size: 11 } } },
                y: { title: { display: true, text: 'Costo Total (USD)', font: { weight: 'bold', size: 11 } } }
            }
        }
    });

    const ctxPie = document.getElementById('costPieChart').getContext('2d');
    if (chartPie) chartPie.destroy();
    chartPie = new Chart(ctxPie, {
        type: 'doughnut',
        data: {
            labels: ['Hormigón', 'Acero'],
            datasets: [{
                data: [mejor.costoConcreto, mejor.costoAceroMat],
                backgroundColor: ['#3b82f6', '#f59e0b'],
                borderColor: ['#1e40af', '#b45309'],
                borderWidth: 2
            }]
        },
        plugins: [whiteBgPlugin],
        options: {
            responsive: true, maintainAspectRatio: false,
            plugins: {
                legend: { position: 'bottom', labels: { usePointStyle: true, padding: 12, font: { size: 11 } } },
                tooltip: { callbacks: { label: ctx => { const total = ctx.dataset.data.reduce((a, b) => a + b, 0); if (total === 0) return `${ctx.label}: $0`; const pct = ((ctx.parsed / total) * 100).toFixed(1); return `${ctx.label}: $${ctx.parsed.toFixed(0)} (${pct}%)`; } } }
            }
        }
    });
}

function cambiarCriterio(valor) {
    criterioActual = valor;
    actualizarStatusInfo();
    if (document.getElementById('resultadosContainer').style.display !== 'none' &&
        document.getElementById('resultadosContainer').innerHTML.trim() !== '') {

function calcular() {
    setStatus('Optimizando...', '');
    setTimeout(() => {
        const resultado = optimizar();
        if (resultado.error) {
            const c = document.getElementById('resultadosContainer');
            c.innerHTML = `<div class="diag-box"><h3>⚠️ Parámetros incorrectos</h3><p>${resultado.error}</p></div>`;
            c.style.display = 'block';
            document.getElementById('resultadosVacio').style.display = 'none';
            document.getElementById('chartsSection').style.display = 'none';
            document.getElementById('chartsVacio').style.display = 'block';
            setStatus('Revisa los parámetros', 'fail');
            switchTab('resultados');
            return;
        }
        if (!resultado.mejor) return;
        ultimoResultado = resultado;
        renderizarResultados(resultado);
        switchTab('resultados');
    }, 100);
}

                <li>La <b>robustez</b> se define como el margen normalizado entre ρ y ρ_max; mayor valor indica mayor reserva de ductilidad.</li>
                <li>La <b>frontera de Pareto</b> agrupa secciones donde ninguna otra mejora simultáneamente en costo y robustez.</li>
                <li>El armado propuesto es orientativo; verificar cuantías mínimas/máximas, separación entre barras y recubrimientos según norma.</li>
                <li>Los gráficos corresponden a la sección evaluada al momento de generar este informe.</li>
            </ul>
        </div>

        <div class="page-foot"><span>${esc(elemento)}</span><span>Página 3</span></div>
    </div>`;

    document.getElementById('reporte').innerHTML = caratula + memoria + anexo;
    document.getElementById('rtSub').textContent = `${norma} · ${elemento}`;
    document.getElementById('reportOverlay').classList.add('open');
    if (imprimir) setTimeout(() => window.print(), 200);
}
