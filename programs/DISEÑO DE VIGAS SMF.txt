const Calc = {
    calcularAs(Mu, b, d, fc, fy, phi){
        if (Mu <= 0) return 0;
        const Mu_kgfcm = Mu * 100000;
        const A = (fy * fy) / (2 * 0.85 * fc * b);
        const B = -fy * d;
        const C = Mu_kgfcm / phi;
        const disc = B * B - 4 * A * C;
        if (disc < 0) return null;
        const As = (-B - Math.sqrt(disc)) / (2 * A);
        const As_min_1 = (14 / fy) * b * d;
        const As_min_2 = (0.8 * Math.sqrt(fc) / fy) * b * d;
        const As_min = Math.max(As_min_1, As_min_2);
        const beta1 = fc <= 280 ? 0.85 : Math.max(0.85 - 0.05 * (fc - 280) / 70, 0.65);
        const cuantia_max = 0.85 * beta1 * (fc / fy) * (0.003 / (0.003 + 0.005));
        const As_max = cuantia_max * b * d;
        return Math.min(Math.max(As, As_min, 0), As_max);
    },
    calc_a_c(As, fy, fc, b, beta1){
        if (As <= 0) return {a:0, c:0};
        const a = (As * fy) / (0.85 * fc * b);
        return {a, c: a / beta1};
    },
    calcularMpr(As, fy, fc, b, d, alpha){
        if (As <= 0) return 0;
        const a = (alpha * fy * As) / (0.85 * fc * b);
        return alpha * fy * As * (d - a / 2) / 100000;
    },
    verificarVc(Vp, Vd, Pu, Ag, fc){
        const cond1 = Vp / Vd >= 0.5;
        const Pc = (Ag * fc) / 20;
        const cond2 = Pu * 1000 <= Pc;
        return {Ra: cond1 ? 0 : 1, Rb: cond2 ? 0 : 1, Pc, aplicarVc: (cond1 ? 0 : 1) + (cond2 ? 0 : 1) >= 1};
    },
    calcularVc(Pu, Ag, fc, b, d){
        const factor = 1 + (Pu * 1000) / (140 * Ag);
        return 0.53 * factor * Math.sqrt(fc) * b * d / 1000;
    },
    calcularSreq(Av, fy, d, Vs){
        if (Vs <= 0) return Infinity;
        return (Av * fy * d) / (Vs * 1000);
    }
};

function dibujarEsquemaViga(canvasId, resultadosTramos, bViga, hViga){
    const canvas = document.getElementById(canvasId);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const n = resultadosTramos.length;

    const MARGIN = 60, APOYO_W = 42, COLUMN_DRAW_W = 26, PX_PER_M = 54, TRAMO_MIN_W = 200,
          INSET = 3, BEAM_H = 46, COL_H = 138, TOP_PAD = 150, BOTTOM_PAD = 140;
    const FONT_TITLE1 = "600 13px 'IBM Plex Sans'", FONT_TITLE2 = "11px 'IBM Plex Mono'",
          FONT_COTA_TOTAL = "600 12.5px 'IBM Plex Mono'", FONT_COTA_TRAMO = "12px 'IBM Plex Mono'",
          FONT_LABEL = "11px 'IBM Plex Mono'", FONT_APOYO = "600 11.5px 'IBM Plex Sans'",
          FONT_ESTRIBO = "600 10.5px 'IBM Plex Mono'", FONT_RAMAS = "600 10.5px 'IBM Plex Mono'";

    const apoyos = [], tramos = [];
    let x = MARGIN;
    apoyos.push({left:x, right:x+APOYO_W, center:x+APOYO_W/2});
    x += APOYO_W;
    for (let i = 0; i < n; i++){
        const w = Math.max(TRAMO_MIN_W, resultadosTramos[i].Ln * PX_PER_M);
        tramos.push({start:x, end:x+w, width:w});
        x += w;
        apoyos.push({left:x, right:x+APOYO_W, center:x+APOYO_W/2});
        x += APOYO_W;

    const canvasWidth = x + MARGIN;

    canvas.style.width = canvasWidth + 'px'; canvas.style.height = canvasHeight + 'px';

function drawCota(ctx, x1, x2, y){
    ctx.beginPath(); ctx.moveTo(x1, y); ctx.lineTo(x2, y); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x1, y-5); ctx.lineTo(x1, y+5); ctx.moveTo(x2, y-5); ctx.lineTo(x2, y+5); ctx.stroke();
}

/* ================================================================
   EXPORTACIÓN DXF — R12 (AC1009) 100% COMPATIBLE
   Solo LINE, TEXT, CIRCLE y SOLID. Sin LWPOLYLINE (no existe en R12).
   Todas las capas usan CONTINUOUS (definido en LTYPE).
   ================================================================ */
function generarDXF(rt, S) {
    const b = S.b, h = S.h, REC = S.r || 2.5;
    const ACOL = 50;             // ancho de columna = 0.5 m
    const COL_SUP = 100;         // 1 m por encima de la viga
    const COL_INF = 100;         // 1 m por debajo de la viga
    const n = rt.length;

    const dxf = [];
    const push = (...vals) => vals.forEach(v => dxf.push(String(v)));
    const f2 = v => Number(v).toFixed(2);

    // Ejes de apoyo (centro de cada columna)
    const xc = [ACOL / 2];
    for (let i = 1; i <= n; i++) xc[i] = xc[i-1] + ACOL + rt[i-1].Ln * 100;
    const TW = xc[n] + ACOL / 2;

    // Cotas verticales
    const yPos = REC + 1.1;
    const yNeg = h - (REC + 1.1);
    const yColSup = h + COL_SUP;
    const yColInf = -COL_INF;
    const yEjeSup = yColSup + 35;

    /* ================= HEADER ================= */
    push('0','SECTION','2','HEADER');
    push('9','$ACADVER','1','AC1009');
    push('9','$INSBASE','10','0.0','20','0.0','30','0.0');
    push('9','$EXTMIN','10','0.0','20',f2(yColInf - 60),'30','0.0');
    push('9','$EXTMAX','10',f2(TW + 60),'20',f2(yEjeSup + 60),'30','0.0');
    push('9','$INSUNITS','70','4');
    push('9','$MEASUREMENT','70','1');
    push('9','$LTSCALE','40','1.0');
    push('9','$TEXTSTYLE','7','STANDARD');
    push('9','$CLAYER','8','0');
    push('0','ENDSEC');

    /* ================= TABLES ================= */
    push('0','SECTION','2','TABLES');

    push('0','TABLE','2','LTYPE','70','1');
    push('0','LTYPE','2','CONTINUOUS','70','64','3','Solid line','72','65','73','0','40','0.0');
    push('0','ENDTAB');

    push('0','TABLE','2','LAYER','70','8');
    const layers = [
        ['0',7], ['VIGAS',7], ['FIERROS',6], ['EJES',2],
        ['TEXTOS',1], ['COTAS',3], ['SIMBOLOS',4], ['COLUMNAS',8]
    ];
    layers.forEach(([name,color]) => {
        push('0','LAYER','2',name,'70','0','62',String(color),'6','CONTINUOUS');
    });
    push('0','ENDTAB');

    push('0','TABLE','2','STYLE','70','1');
    push('0','STYLE','2','STANDARD','70','0','40','0.0','41','1.0','50','0.0','71','0','42','0.2','3','txt','4','');
    push('0','ENDTAB');

    push('0','ENDSEC');

    /* ================= ENTITIES ================= */
    push('0','SECTION','2','ENTITIES');

    // --- helpers ---
    const addLine = (x1,y1,x2,y2,layer) => {
        push('0','LINE','8',layer,
             '10',f2(x1),'20',f2(y1),'30','0.0',
             '11',f2(x2),'21',f2(y2),'31','0.0');
    };
    const addCircle = (x,y,r,layer) => {
        push('0','CIRCLE','8',layer,'10',f2(x),'20',f2(y),'30','0.0','40',f2(r));
    };
    // TEXT centrado (72=1 center, 73=2 middle) — para el texto dentro del círculo
    const addTextC = (x,y,text,height,layer) => {
        push('0','TEXT','8',layer,
             '10',f2(x),'20',f2(y),'30','0.0',
             '40',f2(height),'1',String(text),
             '50','0.0','7','STANDARD',
             '72','1','73','2',
             '11',f2(x),'21',f2(y),'31','0.0');
    };
    // TEXT alineado a la izquierda (esquina inferior izquierda)
    const addText = (x,y,text,height,layer) => {
        push('0','TEXT','8',layer,
             '10',f2(x),'20',f2(y),'30','0.0',
             '40',f2(height),'1',String(text),
             '50','0.0','7','STANDARD');
    };

    /* ---- 1. VIGA + COLUMNAS como UN SOLO contorno continuo ----
       La viga va de x=0 a x=TW, y de y=0 a y=h.
       Las columnas van de y=h hacia arriba y de y=0 hacia abajo.
       En las intersecciones NO se dibujan líneas (se cortan para
       que se vea como una sola pieza monolítica).                      */

    // Viga: cara superior e inferior segmentadas entre columnas
    for (let j = 0; j < n; j++) {
        const x1 = xc[j] + ACOL/2;      // borde derecho de la columna j
        const x2 = xc[j+1] - ACOL/2;    // borde izquierdo de la columna j+1
        addLine(x1, h, x2, h, 'VIGAS'); // cara superior de la viga
        addLine(x1, 0, x2, 0, 'VIGAS'); // cara inferior de la viga
    }
    // Caras laterales de la viga (coinciden con las caras de las columnas extremas)
    addLine(0, 0, 0, h, 'VIGAS');
    addLine(TW, 0, TW, h, 'VIGAS');

    // Columnas: solo 3 lados (arriba/abajo), sin base/tope porque coinciden con la viga
    for (let j = 0; j <= n; j++) {
        const cx = xc[j];
        const cxl = cx - ACOL/2;
        const cxr = cx + ACOL/2;

        // Columna superior
        addLine(cxl, h, cxl, yColSup, 'COLUMNAS');
        addLine(cxr, h, cxr, yColSup, 'COLUMNAS');
        addLine(cxl, yColSup, cxr, yColSup, 'COLUMNAS');

        // Columna inferior
        addLine(cxl, yColInf, cxl, 0, 'COLUMNAS');
        addLine(cxr, yColInf, cxr, 0, 'COLUMNAS');
        addLine(cxl, yColInf, cxr, yColInf, 'COLUMNAS');
    }

    /* ---- 2. EJES verticales con círculo envolviendo el texto ---- */
    for (let j = 0; j <= n; j++) {
        addLine(xc[j], yColInf - 20, xc[j], yEjeSup - 14, 'EJES');
        addCircle(xc[j], yEjeSup, 14, 'SIMBOLOS');
        addTextC(xc[j], yEjeSup, 'E' + j, 12, 'TEXTOS');
    }

    /* ---- 3. ACERO SUPERIOR: de extremo a extremo con ganchos hacia abajo ---- */
    addLine(REC, yNeg, TW - REC, yNeg, 'FIERROS');
    addLine(REC, yNeg, REC, yNeg - 20, 'FIERROS');
    addLine(TW - REC, yNeg, TW - REC, yNeg - 20, 'FIERROS');

    /* ---- 4. ACERO INFERIOR: de extremo a extremo con ganchos hacia arriba ---- */
    addLine(REC, yPos, TW - REC, yPos, 'FIERROS');
    addLine(REC, yPos, REC, yPos + 20, 'FIERROS');
    addLine(TW - REC, yPos, TW - REC, yPos + 20, 'FIERROS');

    /* ---- 5. At y Ab en izq, centro y der de cada tramo ---- */
    rt.forEach((r, i) => {
        const xIzq = xc[i] + ACOL/2 + 8;
        const xDer = xc[i+1] - ACOL/2 - 60;
        const xCen = (xc[i] + xc[i+1]) / 2 - 24;

        const yAt = h + 8;    // justo encima de la viga (entre columnas)
        const yAb = -16;      // justo debajo de la viga

        // At (acero superior)
        addText(xIzq, yAt, 'At=' + r.As_sup_izq.toFixed(2), 8, 'TEXTOS');
        addText(xCen, yAt, 'At=' + r.As_sup_cen.toFixed(2), 8, 'TEXTOS');
        addText(xDer, yAt, 'At=' + r.As_sup_der.toFixed(2), 8, 'TEXTOS');

        // Ab (acero inferior)
        addText(xIzq, yAb, 'Ab=' + r.As_inf_izq.toFixed(2), 8, 'TEXTOS');
        addText(xCen, yAb, 'Ab=' + r.As_inf_cen.toFixed(2), 8, 'TEXTOS');
        addText(xDer, yAb, 'Ab=' + r.As_inf_der.toFixed(2), 8, 'TEXTOS');
    });

    /* ---- 6. Estribos (rótulos) ---- */
    rt.forEach((r, i) => {
        const fL = xc[i] + ACOL/2;
        const fR = xc[i+1] - ACOL/2;
        const txtIzq = 'EST ' + (r.Nramas_izq > 2 ? 'DOBLE ' : '') + 'D' + r.diam_izq + ' c/' + Math.round(r.Sdef_izq);
        const txtDer = 'EST ' + (r.Nramas_der > 2 ? 'DOBLE ' : '') + 'D' + r.diam_der + ' c/' + Math.round(r.Sdef_der);
        addText(fL + 10, -45, txtIzq, 8, 'TEXTOS');
        addText(fR - 100, -45, txtDer, 8, 'TEXTOS');
    });

    /* ---- 7. Longitud de confinamiento a izq y der de cada tramo ---- */
    rt.forEach((r, i) => {
        const xL = xc[i] + ACOL/2 + 10;
        const xR = xc[i+1] - ACOL/2 - 70;
        addText(xL, h + 25, 'Lconf=' + (r.Lconf_izq/100).toFixed(2) + 'm', 8, 'TEXTOS');
        addText(xR, h + 25, 'Lconf=' + (r.Lconf_der/100).toFixed(2) + 'm', 8, 'TEXTOS');
    });

    /* ---- 8. Cotas por tramo (sin cota de longitud total) ---- */
    rt.forEach((r, i) => {
        const fL = xc[i] + ACOL/2;
        const fR = xc[i+1] - ACOL/2;
        const yD = yColSup + 30;
        addLine(fL, yD, fR, yD, 'COTAS');
        addLine(fL, yD-3, fL, yD+3, 'COTAS');
        addLine(fR, yD-3, fR, yD+3, 'COTAS');
        addText((fL + fR)/2 - 18, yD + 5, 'L' + (i+1) + '=' + r.Ln.toFixed(2) + 'm', 10, 'COTAS');
    });

    /* ---- 9. Rótulo de viga ---- */
    addText(10, yColSup + 100, 'VIGA ' + b + ' x ' + h + ' cm', 12, 'TEXTOS');

    push('0','ENDSEC');
    push('0','EOF');

    return dxf.join('\n');
}

function calcularTodo(){
    const h = parseFloat(document.getElementById('h').value);
    const b = parseFloat(document.getElementById('b').value);
    const r = parseFloat(document.getElementById('r').value);
    const d_sup = parseFloat(document.getElementById('d_sup').value);
    const fc = parseFloat(document.getElementById('fc').value);
    const fy = parseFloat(document.getElementById('fy').value);
    const n = parseInt(document.getElementById('nTramos').value) || 1;
    const Pu = 0;

    if (isNaN(h) || isNaN(b) || isNaN(r) || isNaN(fc) || isNaN(fy)){
        alert('Complete los parámetros globales (geometría y materiales).');
        return;
    }

    const Ln = [];
    for (let i = 1; i <= n; i++) Ln.push(parseFloat(document.getElementById(`Ln_${i}`).value) || 1);

    const M_neg_apoyo = [], M_pos_apoyo = [], M_neg_centro = [], M_pos_centro = [];
    for (let i = 0; i <= n; i++){
        M_neg_apoyo[i] = parseFloat(document.getElementById(`M_neg_apoyo_${i}`).value) || 0;
        M_pos_apoyo[i] = parseFloat(document.getElementById(`M_pos_apoyo_${i}`).value) || 0;
    }
    for (let i = 1; i <= n; i++){
        M_neg_centro[i] = parseFloat(document.getElementById(`M_neg_centro_${i}`).value) || 0;
        M_pos_centro[i] = parseFloat(document.getElementById(`M_pos_centro_${i}`).value) || 0;
    }

    const Vu_izq = [], Vu_der = [], Vg = [];
    for (let i = 1; i <= n; i++){
        Vu_izq.push(parseFloat(document.getElementById(`Vu_izq_${i}`).value) || 0);
        Vu_der.push(parseFloat(document.getElementById(`Vu_der_${i}`).value) || 0);
        Vg.push(parseFloat(document.getElementById(`Vg_${i}`).value) || 0);
    }

    const estribosExtremos = [];
    for (let i = 1; i <= n; i++){
        estribosExtremos.push({tramo:i, extremo:'izq',
            diam: parseFloat(document.getElementById(`diam_izq_${i}`).value) || 10,
            ramas: parseInt(document.getElementById(`ramas_izq_${i}`).value) || 2});
        estribosExtremos.push({tramo:i, extremo:'der',
            diam: parseFloat(document.getElementById(`diam_der_${i}`).value) || 10,
            ramas: parseInt(document.getElementById(`ramas_der_${i}`).value) || 2});
    }

    const beta1 = fc <= 280 ? 0.85 : Math.max(0.85 - 0.05 * (fc - 280) / 70, 0.65);
    const phi_b = 0.9, phi_v = 0.6, alpha = 1.25;
    const d = h - r;
    const Ag = b * h;
    const db_long = 2.0;

    const As_min_1 = (14 / fy) * b * d;
    const As_min_2 = (0.8 * Math.sqrt(fc) / fy) * b * d;
    const As_min = Math.max(As_min_1, As_min_2);

    const As_req_neg_apoyo = new Array(n+1).fill(0);
    const As_req_pos_apoyo = new Array(n+1).fill(0);
    const As_req_neg_centro = new Array(n+1).fill(0);
    const As_req_pos_centro = new Array(n+1).fill(0);

    for (let i = 0; i <= n; i++){
        As_req_neg_apoyo[i] = Calc.calcularAs(Math.abs(M_neg_apoyo[i]), b, d, fc, fy, phi_b);
        As_req_pos_apoyo[i] = Calc.calcularAs(Math.abs(M_pos_apoyo[i]), b, d, fc, fy, phi_b);
    }
    for (let i = 1; i <= n; i++){
        As_req_neg_centro[i] = Calc.calcularAs(Math.abs(M_neg_centro[i]), b, d, fc, fy, phi_b);
        As_req_pos_centro[i] = Calc.calcularAs(Math.abs(M_pos_centro[i]), b, d, fc, fy, phi_b);
    }

    let maxAsNegApoyo = 0;
    for (let i = 0; i <= n; i++) if (As_req_neg_apoyo[i] > maxAsNegApoyo) maxAsNegApoyo = As_req_neg_apoyo[i];

    const resultadosTramos = [];

    for (let i = 1; i <= n; i++){
        const Ln_i = Ln[i-1];
        const Vg_i = Vg[i-1];
        const Vu_izq_i = Vu_izq[i-1];
        const Vu_der_i = Vu_der[i-1];

        const apoyoIzqIdx = i-1, apoyoDerIdx = i, centroIdx = i;

        let As_sup_izq_req = As_req_neg_apoyo[apoyoIzqIdx];
        let As_inf_izq_req = As_req_pos_apoyo[apoyoIzqIdx];
        let As_sup_cen_req = As_req_neg_centro[centroIdx];
        let As_inf_cen_req = As_req_pos_centro[centroIdx];
        let As_sup_der_req = As_req_neg_apoyo[apoyoDerIdx];
        let As_inf_der_req = As_req_pos_apoyo[apoyoDerIdx];

        if (As_inf_izq_req < 0.5 * As_sup_izq_req) As_inf_izq_req = 0.5 * As_sup_izq_req;
        if (As_inf_der_req < 0.5 * As_sup_der_req) As_inf_der_req = 0.5 * As_sup_der_req;

        const minCentro = 0.25 * maxAsNegApoyo;
        if (As_sup_cen_req < minCentro) As_sup_cen_req = minCentro;
        if (As_inf_cen_req < minCentro) As_inf_cen_req = minCentro;

        As_sup_izq_req = Math.max(As_sup_izq_req, As_min);
        As_inf_izq_req = Math.max(As_inf_izq_req, As_min);
        As_sup_cen_req = Math.max(As_sup_cen_req, As_min);
        As_inf_cen_req = Math.max(As_inf_cen_req, As_min);
        As_sup_der_req = Math.max(As_sup_der_req, As_min);
        As_inf_der_req = Math.max(As_inf_der_req, As_min);

        const estIzq = estribosExtremos.find(e => e.tramo === i && e.extremo === 'izq');
        const estDer = estribosExtremos.find(e => e.tramo === i && e.extremo === 'der');
        const diam_izq = estIzq ? estIzq.diam : 10;
        const diam_der = estDer ? estDer.diam : 10;
        const Nramas_izq = estIzq ? estIzq.ramas : 2;
        const Nramas_der = estDer ? estDer.ramas : 2;

        const a_neg_izq = Calc.calc_a_c(As_sup_izq_req, fy, fc, b, beta1).a;
        const a_pos_cen = Calc.calc_a_c(As_inf_cen_req, fy, fc, b, beta1).a;
        const a_neg_der = Calc.calc_a_c(As_sup_der_req, fy, fc, b, beta1).a;

        const Mpr_izq = Calc.calcularMpr(As_sup_izq_req, fy, fc, b, d, alpha);
        const Mpr_der = Calc.calcularMpr(As_sup_der_req, fy, fc, b, d, alpha);
        const Vp = (Mpr_izq + Mpr_der) / Ln_i;
        const Ve_izq = Vg_i + Vp;
        const Ve_der = Vg_i - Vp;
        const Vd_izq = Math.max(Math.abs(Ve_izq), Math.abs(Vu_izq_i));
        const Vd_der = Math.max(Math.abs(Ve_der), Math.abs(Vu_der_i));

        const resIzq = Calc.verificarVc(Vp, Vd_izq, Pu, Ag, fc);
        const Vc_izq = resIzq.aplicarVc ? Calc.calcularVc(Pu, Ag, fc, b, d) : 0;
        const Vs_izq = Math.max((Vd_izq / phi_v) - Vc_izq, 0);

        const resDer = Calc.verificarVc(Vp, Vd_der, Pu, Ag, fc);
        const Vc_der = resDer.aplicarVc ? Calc.calcularVc(Pu, Ag, fc, b, d) : 0;
        const Vs_der = Math.max((Vd_der / phi_v) - Vc_der, 0);

        const area_estribo_izq = Math.PI/4 * Math.pow(diam_izq/10, 2);
        const Av_izq = Nramas_izq * area_estribo_izq;
        const Sreq_izq = Calc.calcularSreq(Av_izq, fy, d, Vs_izq);

        const area_estribo_der = Math.PI/4 * Math.pow(diam_der/10, 2);
        const Av_der = Nramas_der * area_estribo_der;
        const Sreq_der = Calc.calcularSreq(Av_der, fy, d, Vs_der);

        const s1 = d/4, s2 = 6*db_long, s3 = 15;
        const smax_norma = Math.min(s1, s2, s3);
        const Sdef_izq = Math.min(Sreq_izq, smax_norma);
        const Sdef_der = Math.min(Sreq_der, smax_norma);

        const Lconf_izq = Math.max(2*h, 40);
        const Lconf_der = Math.max(2*h, 40);

        resultadosTramos.push({
            tramo:i, Ln:Ln_i,
            M_neg_izq:M_neg_apoyo[apoyoIzqIdx], M_pos_izq:M_pos_apoyo[apoyoIzqIdx],
            M_neg_cen:M_neg_centro[centroIdx], M_pos_cen:M_pos_centro[centroIdx],
            M_neg_der:M_neg_apoyo[apoyoDerIdx], M_pos_der:M_pos_apoyo[apoyoDerIdx],
            As_sup_izq:As_sup_izq_req, As_inf_izq:As_inf_izq_req,
            As_sup_cen:As_sup_cen_req, As_inf_cen:As_inf_cen_req,
            As_sup_der:As_sup_der_req, As_inf_der:As_inf_der_req,
            a_neg_izq, a_pos_cen, a_neg_der,
            Mpr_izq, Mpr_der, Vp, Ve_izq, Ve_der, Vd_izq, Vd_der,
            Vc_izq, Vc_der, Vs_izq, Vs_der, Sreq_izq, Sreq_der, Sdef_izq, Sdef_der,
            Lconf_izq, Lconf_der,
            Ra_izq:resIzq.Ra, Rb_izq:resIzq.Rb, Ra_der:resDer.Ra, Rb_der:resDer.Rb,
            diam_izq, diam_der, Nramas_izq, Nramas_der, smax_norma,
            Vg:Vg_i, Vu_izq:Vu_izq_i, Vu_der:Vu_der_i
        });
    }

    ultimoResultadosTramos = resultadosTramos;
    ultimoBH = {b, h};
    ultimoResumen = {
        h, b, r, d_sup, fc, fy, n, d, Ag, As_min,
        phi_b, phi_v, alpha, beta1,
        proyecto: document.getElementById('proyNombre').value || '—',
        elemento: document.getElementById('proyElemento').value || '—',
        norma: document.getElementById('proyNorma').value || 'ACI 318 · SMF',
        ubicacion: document.getElementById('proyUbicacion').value || '—'
    };

    renderResultados(resultadosTramos, ultimoResumen);
    dibujarEsquemaViga('esquemaVigaCanvas', resultadosTramos, b, h);
    const ev = document.getElementById('esquemaVacio');
    if (ev) ev.style.display = 'none';
    actualizarBarraEstado(true);
    switchTab('resultados');
}

}
