/* Sección transversal con el estribo (énfasis en el refuerzo transversal) */
function dibujarSeccionEstribo(canvas, b_cm, h_cm, recub_cm, diamEst, nRam, sCm, d_cm, estado) {
    if (typeof canvas === 'string') canvas = document.getElementById(canvas);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const escala = 7;
    const mSup = 72, mInf = 60, mIzq = 55, mDer = 55;
    const ancho = b_cm * escala + mIzq + mDer;
    const alto = h_cm * escala + mSup + mInf;
    canvas.width = ancho;
    canvas.height = alto;
    canvas.style.width = ancho + 'px';
    canvas.style.height = alto + 'px';
    ctx.clearRect(0, 0, ancho, alto);

    const x0 = mIzq, y0 = mSup;
    const fw = b_cm * escala, fh = h_cm * escala;

    // Encabezado
    ctx.fillStyle = '#1e293b'; ctx.font = 'bold 13px IBM Plex Sans'; ctx.textAlign = 'center';
    ctx.fillText('SECCIÓN TRANSVERSAL · REFUERZO CORTANTE', ancho/2, 24);
    ctx.font = '10px IBM Plex Mono'; ctx.fillStyle = '#5c6b7a';
    ctx.fillText(`b = ${b_cm} cm × h = ${h_cm} cm · r = ${recub_cm} cm`, ancho/2, 40);

    // Fondo de la sección
    ctx.fillStyle = '#f2f5f8';
    ctx.fillRect(x0, y0, fw, fh);
    ctx.strokeStyle = '#6d7a88'; ctx.lineWidth = 2;
    ctx.strokeRect(x0, y0, fw, fh);

    // Líneas guía del recubrimiento
    const rPix = recub_cm * escala;
    ctx.strokeStyle = '#c8d1d8'; ctx.lineWidth = 1; ctx.setLineDash([4,3]);
    ctx.strokeRect(x0 + rPix, y0 + rPix, fw - 2*rPix, fh - 2*rPix);
    ctx.setLineDash([]);

    // Barras longitudinales esquemáticas en las esquinas del estribo
    ctx.fillStyle = '#0e6cb3';
    const dbPix = 5;
    [[x0+rPix, y0+rPix],[x0+fw-rPix, y0+rPix],[x0+rPix, y0+fh-rPix],[x0+fw-rPix, y0+fh-rPix]].forEach(p => {
        ctx.beginPath(); ctx.arc(p[0], p[1], dbPix, 0, Math.PI*2); ctx.fill();
    });

    // Estribo cerrado (perímetro del recubrimiento): ganchos 135°
    ctx.strokeStyle = '#ca244b'; ctx.lineWidth = 3;
    ctx.strokeRect(x0 + rPix, y0 + rPix, fw - 2*rPix, fh - 2*rPix);
    // Gancho en una esquina (135°)
    ctx.strokeStyle = '#ca244b'; ctx.lineWidth = 3;
    const gx = x0 + fw - rPix, gy = y0 + rPix;
    ctx.beginPath();
    ctx.moveTo(gx - 10, gy);
    ctx.lineTo(gx, gy - 9 + 16);
    ctx.stroke();
    ctx.font = 'bold 9px IBM Plex Mono'; ctx.fillStyle = '#ca244b';
    ctx.fillText('135°', gx - 40, gy - 14);

    // Ramales verticales del estribo (2, 4 o más)
    const nR = Math.max(1, Math.min(nRam, 6));
    ctx.strokeStyle = '#ca244b'; ctx.lineWidth = 3;
    for (let i = 1; i < nR; i++){
        const rel = i / nR;
        const rx = x0 + rPix + rel * (fw - 2*rPix);
        ctx.beginPath(); ctx.moveTo(rx, y0 + rPix); ctx.lineTo(rx, y0 + fh - rPix); ctx.stroke();
    }
    // Rótulo de ramales (parte superior entre ramal exterior)
    ctx.fillStyle = '#33404d'; ctx.font = 'bold 10px IBM Plex Mono';
    ctx.fillText(`${nR} ramales · φ${diamEst} mm`, x0 + rPix + 4, y0 - 10);

    // Cota de ancho b
    ctx.strokeStyle = '#596775'; ctx.fillStyle = '#3f4c59'; ctx.lineWidth = 1;
    ctx.font = '10px IBM Plex Mono'; ctx.textAlign = 'center';
    ctx.beginPath();
    ctx.moveTo(x0, y0 + fh + 24);
    ctx.lineTo(x0 + fw, y0 + fh + 24);
    ctx.stroke();
    dibujarFlecha(ctx, x0, y0+fh+24, x0+fw, y0+fh+24);
    ctx.fillText(`b = ${b_cm} cm`, x0 + fw/2, y0 + fh + 38);

    // Cota de altura h (lado derecho)
    const dirM = Math.max(nR,2);
    const cxRight = x0 + fw + 26;
    ctx.beginPath(); ctx.moveTo(cxRight, y0); ctx.lineTo(cxRight, y0+fh); ctx.stroke();
    ctx.save(); ctx.translate(cxRight + 16, y0 + fh/2); ctx.rotate(-Math.PI/2);
    ctx.fillText(`h = ${h_cm} cm`, 0, 0); ctx.restore();

    // Cota de altura efectiva d (lado izquierdo, desde abajo)
    const cxLeft = x0 - 26;
    const dPix = (h_cm - recub_cm) * escala;
    ctx.beginPath(); ctx.moveTo(cxLeft, y0 + fh - dPix); ctx.lineTo(cxLeft, y0+fh); ctx.stroke();
    ctx.save(); ctx.translate(cxLeft - 16, y0 + fh - dPix/2); ctx.rotate(-Math.PI/2);
    ctx.fillText(`d = ${d_cm.toFixed(1)} cm`, 0, 0); ctx.restore();

    // Leyenda inferior
    const estadoColor = estado === 'ok' ? '#12803f' : estado === 'fail' ? '#bf2a2a' : estado === 'warn' ? '#97600a' : '#1e40af';
    ctx.fillStyle = estadoColor; ctx.font = 'bold 10px IBM Plex Mono'; ctx.textAlign = 'left';
    ctx.fillText(`● Estribo φ${diamEst} mm @ ${sCm}`, x0, y0 + fh + 52);
    ctx.font = '9.5px IBM Plex Mono'; ctx.fillStyle = '#33404d';
    ctx.fillText('Cerco cerrado con ganchos a 135° según ACI 318', x0, y0 + fh + 66);
}

function dibujarFlecha(ctx, x1, y1, x2, y2){
    const ang = Math.atan2(y2-y1, x2-x1), s = 7;
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(x2,y2);
    ctx.lineTo(x2 - s*Math.cos(ang-0.4), y2 - s*Math.sin(ang-0.4));
    ctx.moveTo(x2,y2);
    ctx.lineTo(x2 - s*Math.cos(ang+0.4), y2 - s*Math.sin(ang+0.4));
    ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x1,y1);
    ctx.lineTo(x1 + s*Math.cos(ang-0.4), y1 + s*Math.sin(ang-0.4));
    ctx.moveTo(x1,y1);
    ctx.lineTo(x1 + s*Math.cos(ang+0.4), y1 + s*Math.sin(ang+0.4));
    ctx.stroke();
}

let ultimoResultado = null;

function calcularTodo() {
    const b_cm = parseFloat(document.getElementById('b').value) || 30;
    const h_cm = parseFloat(document.getElementById('h').value) || 50;
    const recub_cm = parseFloat(document.getElementById('recub').value) || 6;
    const fc = parseFloat(document.getElementById('fc').value) || 25;
    let fy = parseFloat(document.getElementById('fy').value) || 420;
    const VuKN = parseFloat(document.getElementById('Vu').value) || 0;
    const fimm = parseFloat(document.getElementById('diamEstribo').value) || 8;
    const noRamales = parseInt(document.getElementById('ramales').value) || 2;

    if ([b_cm, h_cm, recub_cm, fc, fy, VuKN].some(isNaN)) {
        alert('Complete todos los campos.');
        return;
    }
    if (fy > 420) fy = 420;

    const b = b_cm/100, h = h_cm/100, recub_eje = recub_cm/100;
    const d = h - recub_eje;
    const Av = noRamales * (Math.PI * (fimm/1000)**2 / 4);
    const Av1ramal = Math.PI * (fimm/1000)**2 / 4;

    // Capacidades (MN)
    const Vc = Math.sqrt(fc) * b * d / 6;
    const fiVc = 0.75 * Vc;
    const VsLimite = (2/3) * Math.sqrt(fc) * b * d;
    const fiVnMax = 0.75 * (Vc + VsLimite);

    const Vu = VuKN / 1000;
    const sMaxBaseCm = Math.floor(d/2*100);
    const sConstCm = 60;

    let estado = '', mensaje = '', sFinalCm = '-', avMinCm = '-', noRamalesReq = noRamales;
    let detalleExtra = '', vsReq = 0, sCalcCm = 0, sMaxLocal = 0, avReq = 0, sCalc = 0;

    if (Vu > fiVnMax) {
        estado = 'fail';
        mensaje = 'La sección es insuficiente incluso con estribos. Aumente dimensiones o f\'c.';
    } else if (Vu < 0.5*fiVc) {
        estado = 'info';
        mensaje = 'No se requieren estribos por cálculo. Colocar refuerzo constructivo.';
        sFinalCm = sConstCm + ' cm';
    } else if (Vu < fiVc) {
        estado = 'warn';
        const sMax = sMaxBaseCm/100;
        const AvMin = Math.max(0.062*Math.sqrt(fc)*b*sMax/fy, 0.35*b*sMax/fy);
        avMinCm = (AvMin*10000).toFixed(2);
        avReq = AvMin*10000;
        const ramalesNecesarios = Math.ceil(AvMin/Av1ramal);
        noRamalesReq = Math.max(noRamales, ramalesNecesarios);
        sFinalCm = sMaxBaseCm + ' cm (máx.)';
        mensaje = 'Vu < φVc → requiere estribos mínimos.';
        detalleExtra = `A<sub>v, mín</sub> = ${avMinCm} cm² → ${noRamalesReq} ramales de φ${fimm} mm.`;
    } else {
        estado = 'ok';
        const limite = (1/3)*Math.sqrt(fc)*b*d;
        const Vs = (Vu - fiVc)/0.75;
        vsReq = Vs*1000;
        sCalc = (0.75*Av*fy*d)/(Vu - fiVc);
        sCalcCm = Math.floor(sCalc*100);
        sMaxLocal = (Vs > limite) ? Math.floor(d/4*100) : Math.floor(d/2*100);
        const sCm = Math.min(sCalcCm, sMaxLocal);
        const s = sCm/100;
        const fiVs = 0.75*Av*fy*d/s;
        sFinalCm = sCm + ' cm';
        mensaje = 'Diseño de estribos por demanda de cortante.';
        detalleExtra = `φV<sub>n</sub> = ${(fiVc*1000).toFixed(0)} + ${(fiVs*1000).toFixed(0)} = ${((fiVc+fiVs)*1000).toFixed(0)} kN`;
        if (sCalcCm > sMaxLocal){
            detalleExtra += ` · s limitada a ${sMaxLocal} cm (d/${Vs > limite ? '4' : '2'})`;
        }
    }

    const ultimoResultado = {
        b_cm, h_cm, recub_cm, fc, fy, VuKN, fimm, noRamales,
        d_cm: d*100, Av_cm: Av*10000, avReq, avMinCm, noRamalesReq,
        fiVc_kN: fiVc*1000, medioFiVc_kN: 0.5*fiVc*1000, fiVnMax_kN: fiVnMax*1000,
        sMaxBaseCm, sFinalCm, estado, mensaje, detalleExtra, sCalcCm, vsReq, sMaxLocal
    };

    renderizarResultados(ultimoResultado);
    actualizarBarraEstado(true);
    document.getElementById('btnInforme').disabled = false;
    document.getElementById('btnPDF').disabled = false;
    switchTab('resultados');
}

        <tr><td class="left">Límite sin estribos 0.5·φVc</td><td>${R.medioFiVc_kN.toFixed(1)} kN</td><td>—</td><td><span class="badge ${R.VuKN/1000 < R.medioFiVc_kN/1000 ? 'ok' : 'warn'}">${R.VuKN/1000 < R.medioFiVc_kN/1000 ? 'No requiere' : 'Requiere'}</span></td></tr>
        </tbody></table>
        <div class="stirrup-recom">
            <div><div class="lbl">Estado del diseño</div></div>
            <span class="big">${R.mensaje}</span>
            <span>${badgeMap[R.estado]}</span>
        </div>
        </div></div>`;

    // R2 — Diseño de estribos
    html += `<div class="card">
        <div class="card-head"><h2><span class="idx">R2</span> Diseño de estribos</h2><span class="tag">Refuerzo transversal</span></div>
        <div class="card-body">
        <table class="tbl"><thead><tr><th>Parámetro</th><th>Valor</th><th>Observación</th></tr></thead><tbody>
        <tr><td class="left">Altura efectiva d</td><td>${R.d_cm.toFixed(1)} cm</td><td>h − r</td></tr>
        <tr><td class="left">Av disponible</td><td>${R.Av_cm.toFixed(2)} cm²</td><td>${R.noRamales} ramales × φ${R.fimm} mm</td></tr>
        ${R.avReq > 0 ? `<tr><td class="left">Av mínima requerida</td><td>${R.avMinCm} cm²</td><td>${R.noRamalesReq} ramales → φ${R.fimm} mm</td></tr>` : ''}
        ${R.sCalcCm > 0 ? `<tr><td class="left">Separación calculada</td><td>${R.sCalcCm} cm</td><td>${R.vsReq.toFixed(0)} kN (Vs requerido)</td></tr>` : ''}
        <tr><td class="left">Separación máxima normativa</td><td>${R.sMaxBaseCm} cm</td><td>d/2</td></tr>
        </tbody></table>
        <div class="stirrup-recom">
            <div><div class="lbl">Estribo propuesto</div></div>
            <span class="big">φ${R.fimm} mm @ ${R.sFinalCm}</span>
            <span class="badge ${R.estado === 'fail' ? 'fail' : R.estado === 'warn' ? 'warn' : R.estado === 'info' ? 'info' : 'ok'}">${badgeMap[R.estado].replace(/<[^>]+>/g,'')}</span>
        </div>
        ${R.detalleExtra ? `<p class="hint" style="margin-top:8px;">${R.detalleExtra}</p>` : ''}
        </div></div>`;

    // R3 — Datos del estribo
    html += `<div class="card">
        <div class="card-head"><h2><span class="idx">R3</span> Datos del estribo</h2><span class="tag">Resumen</span></div>
        <div class="card-body">
        <table class="tbl"><thead><tr><th>Diámetro</th><th>Ramales</th><th>Av</th><th>Separación final</th><th>Recomendación</th></tr></thead><tbody>
        <tr>
            <td>φ${R.fimm} mm</td>
            <td>${R.noRamalesReq}</td>
            <td>${R.Av_cm.toFixed(2)} cm²</td>
            <td>${R.sFinalCm}</td>
            <td class="left">${R.mensaje}</td>
        </tr>
        </tbody></table>
        </div></div>`;

    container.innerHTML = html;
}

function esc(s){
    return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

function hoyTexto(){
    return new Date().toLocaleDateString('es-BO', { day:'2-digit', month:'long', year:'numeric' });
}
