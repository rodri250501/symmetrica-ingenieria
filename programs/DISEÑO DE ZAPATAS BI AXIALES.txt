function dibujarPlanoZapata(canvasId, L, B, cx, cy, footingType, eccentricEdge) {
    const canvas = document.getElementById(canvasId);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const w = canvas.width, h = canvas.height;
    ctx.clearRect(0, 0, w, h);
    if (L === 0 || B === 0) return;
    const padding = 20;
    const scale = Math.min((w - 2*padding)/L, (h - 2*padding)/B);
    const fw = L*scale, fh = B*scale;
    const fx = (w - fw)/2, fy = (h - fh)/2;

    ctx.strokeStyle = '#9ca3af'; ctx.lineWidth = 2; ctx.strokeRect(fx, fy, fw, fh);
    const cw = cx*scale, ch = cy*scale;
    let colX, colY;
    if (footingType === 'concentric') {
        colX = fx + (fw - cw)/2; colY = fy + (fh - ch)/2;
    } else if (footingType === 'eccentric') {
        if (eccentricEdge === 'L') { colX = fx; colY = fy + (fh - ch)/2; }
        else { colX = fx + (fw - cw)/2; colY = fy; }
    } else {
        colX = fx; colY = fy;
    }
    ctx.fillStyle = 'rgba(129,140,248,0.5)'; ctx.strokeStyle = '#4f46e5';
    ctx.fillRect(colX, colY, cw, ch); ctx.strokeRect(colX, colY, cw, ch);
    ctx.fillStyle = '#1f2937'; ctx.font = '10px IBM Plex Mono'; ctx.textAlign = 'center';
    ctx.fillText(`L = ${L.toFixed(2)} m`, w/2, fy - 5);
    ctx.save(); ctx.translate(fx - 10, h/2); ctx.rotate(-Math.PI/2); ctx.fillText(`B = ${B.toFixed(2)} m`, 0, 0); ctx.restore();
}

function dibujarEsquemaArmado(canvas, L, B, cx, cy, diamX, sepX, diamY, sepY, As_x_req, As_y_req) {
    if (typeof canvas === 'string') canvas = document.getElementById(canvas);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const escala = 110;
    const mSup = 50, mInf = 90, mIzq = 88, mDer = 42;
    const fw = L * escala, fh = B * escala;
    const ancho = fw + mIzq + mDer;
    const alto = fh + mSup + mInf;
    canvas.width = ancho;
    canvas.height = alto;
    canvas.style.width = ancho + 'px';
    canvas.style.height = alto + 'px';
    ctx.clearRect(0, 0, ancho, alto);

    const x0 = mIzq, y0 = mSup;

    const flecha = (x1,y1,x2,y2) => {
        const ang = Math.atan2(y2-y1, x2-x1), s = 7;
        ctx.lineWidth = 1;
        ctx.beginPath(); ctx.moveTo(x2,y2);
        ctx.lineTo(x2 - s*Math.cos(ang-0.4), y2 - s*Math.sin(ang-0.4));
        ctx.moveTo(x2,y2);
        ctx.lineTo(x2 - s*Math.cos(ang+0.4), y2 - s*Math.sin(ang+0.4));
        ctx.stroke();
    };

    // Encabezado
    ctx.fillStyle = '#1e293b'; ctx.font = 'bold 13px IBM Plex Sans'; ctx.textAlign = 'center';
    ctx.fillText('ESQUEMA DE ARMADO · PLANTA', ancho/2, 22);
    ctx.font = '10px IBM Plex Mono'; ctx.fillStyle = '#5c6b7a';
    ctx.fillText(`Escala · ${L} m de largo`, ancho/2, 37);

    // Zapata
    ctx.fillStyle = '#f2f5f8';
    ctx.fillRect(x0, y0, fw, fh);
    ctx.strokeStyle = '#6d7a88'; ctx.lineWidth = 2;
    ctx.strokeRect(x0, y0, fw, fh);

    // Columna
    const colW = cx*escala, colH = cy*escala;
    const colX = x0 + (fw-colW)/2, colY = y0 + (fh-colH)/2;
    ctx.fillStyle = '#d5dee6'; ctx.fillRect(colX, colY, colW, colH);
    ctx.strokeStyle = '#55616e'; ctx.lineWidth = 1.5;
    ctx.strokeRect(colX, colY, colW, colH);
    ctx.fillStyle = '#33404d'; ctx.font = '10px IBM Plex Mono'; ctx.textAlign = 'center';
    ctx.fillText(`${cx.toFixed(2)} × ${cy.toFixed(2)} m`, colX+colW/2, colY+colH/2+3);

    // Parrilla: barras X (verticales) y barras Y (horizontales)
    const nbX = Math.max(Math.round(L*100/sepX), 2);
    const nbY = Math.max(Math.round(B*100/sepY), 2);
    ctx.lineWidth = 1.8;
    ctx.strokeStyle = '#1c72c4';
    for (let i=1; i<=nbX; i++){
        const x = x0 + (i/(nbX+1))*fw;
        ctx.beginPath(); ctx.moveTo(x, y0); ctx.lineTo(x, y0+fh); ctx.stroke();
    }
    ctx.strokeStyle = '#0e9f8f';
    for (let i=1; i<=nbY; i++){
        const y = y0 + (i/(nbY+1))*fh;
        ctx.beginPath(); ctx.moveTo(x0, y); ctx.lineTo(x0+fw, y); ctx.stroke();
    }

    // Cotas
    ctx.fillStyle = '#3f4c59'; ctx.strokeStyle = '#596775'; ctx.lineWidth = 1;
    // Cota L (inferior)
    const cY = y0 + fh + 32;
    ctx.beginPath(); ctx.moveTo(x0, cY); ctx.lineTo(x0+fw, cY); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x0, y0-4); ctx.lineTo(x0, cY);
    ctx.moveTo(x0+fw, y0-4); ctx.lineTo(x0+fw, cY); ctx.stroke();

    ctx.fillText(`L = ${L.toFixed(2)} m`, x0+fw/2, cY-7);
    // Cota B (izquierda, vertical)
    const cX = x0 - 32;
    ctx.beginPath(); ctx.moveTo(cX, y0); ctx.lineTo(cX, y0+fh); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x0, y0); ctx.lineTo(cX, y0);

    flecha(cX, y0, cX, y0+fh); flecha(cX, y0+fh, cX, y0);

function dibujarDiagramaPresiones(canvas, pressureFn, p_max, p_min) {
    if (typeof canvas === 'string') canvas = document.getElementById(canvas);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const w = canvas.width, h = canvas.height;
    ctx.clearRect(0,0,w,h);
    const footingTop = h*0.4, footingHeight = h*0.15, margin = w*0.1, footingWidth = w*0.8;
    ctx.fillStyle = '#d1d5db'; ctx.fillRect(margin, footingTop, footingWidth, footingHeight);
    const colWidth = footingWidth * (document.getElementById('columnaL').value / document.getElementById('zapataL').value);
    ctx.fillStyle = '#9ca3af'; ctx.fillRect(w/2 - colWidth/2, footingTop - h*0.25, colWidth, h*0.25);
    const pressureBase = footingTop + footingHeight;
    const maxDisplayPressure = Math.max(p_max, Math.abs(p_min))*1.2 || 1;
    const maxPressureHeight = h*0.4;
    ctx.fillStyle = 'rgba(59,130,246,0.3)'; ctx.strokeStyle = '#2563eb'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(margin, pressureBase);
    const steps = 100;
    for (let i=0; i<=steps; i++) {
        const x_norm = i/steps;
        const pressure = pressureFn(x_norm*2-1);
        const p_height = (pressure / maxDisplayPressure) * maxPressureHeight;
        ctx.lineTo(margin + x_norm*footingWidth, pressureBase + p_height);
    }
    ctx.lineTo(margin + footingWidth, pressureBase); ctx.closePath(); ctx.fill(); ctx.stroke();
    ctx.fillStyle = '#1e3a8a'; ctx.font = '10px IBM Plex Mono';
    ctx.textAlign = 'left'; ctx.fillText(`${pressureFn(-1).toFixed(0)} kPa`, margin, h-5);
    ctx.textAlign = 'right'; ctx.fillText(`${pressureFn(1).toFixed(0)} kPa`, margin+footingWidth, h-5);
}

    const Msx_total = Msx_applied + Ps*e_geom_y;
    const Msy_total = Msy_applied + Ps*e_geom_x;
    const ex_total = Msy_total / Ps;
    const ey_total = Msx_total / Ps;
    const kern_x = L/6, kern_y = B/6;

    const q_avg_s = Ps / A;
    const rigid_pressures_s = [
        q_avg_s + Msx_total*(B/2)/Iy + Msy_total*(L/2)/Ix,
        q_avg_s + Msx_total*(B/2)/Iy - Msy_total*(L/2)/Ix,
        q_avg_s - Msx_total*(B/2)/Iy - Msy_total*(L/2)/Ix,
        q_avg_s - Msx_total*(B/2)/Iy + Msy_total*(L/2)/Ix
    ];
    let q_max_s, p_min_s, pressureFn;
    if (metodo === 'rigido') {
        q_max_s = Math.max(...rigid_pressures_s);
        p_min_s = Math.min(...rigid_pressures_s);
        pressureFn = (x_norm) => q_avg_s + Msy_total*(x_norm*L/2)/Ix;
    } else if (metodo === 'winkler') {
        q_max_s = Math.max(...rigid_pressures_s) * (soilType === 'sand' ? 1.2 : 1.3);
        p_min_s = 2*q_avg_s - q_max_s;
        pressureFn = (x_norm) => { const x_abs = Math.abs(x_norm); return soilType === 'sand' ? p_min_s + (q_max_s - p_min_s)*(1 - x_abs**2) : p_min_s + (q_max_s - p_min_s)*x_abs**2; };
    } else {
        q_max_s = Math.max(...rigid_pressures_s) * (soilType === 'sand' ? 1.14 : 1.235);
        p_min_s = 2*q_avg_s - q_max_s;
        pressureFn = (x_norm) => { const x_abs = Math.abs(x_norm); return p_min_s + (q_max_s - p_min_s)*(soilType === 'sand' ? (1 - x_abs**1.5) : x_abs**1.5); };
    }

    const asentamiento_s = (q_avg_s / k_s) * 1000;

    const combinations = [
        { Pu:1.4*PD, Mux:1.4*MDx, Muy:1.4*MDy },
        { Pu:1.2*PD+1.6*PL, Mux:1.2*MDx+1.6*MLx, Muy:1.2*MDy+1.6*MLy },
        { Pu:1.2*PD+1.0*PE+1.0*PL, Mux:1.2*MDx+1.0*MEx+1.0*MLx, Muy:1.2*MDy+1.0*MEy+1.0*MLy },
        { Pu:0.9*PD+1.0*PE, Mux:0.9*MDx+1.0*MEx, Muy:0.9*MDy+1.0*MEy }
    ];

    let max_Mux = 0, max_Muy = 0, max_Vx = 0, max_Vy = 0, max_Vp = 0;
    combinations.forEach(c => {
        if (c.Pu < 0) return;
        const Mux_total = c.Mux + c.Pu*e_geom_y;
        const Muy_total = c.Muy + c.Pu*e_geom_x;
        const q_avg_u = c.Pu / A;
        const pf = (x,y) => q_avg_u + Mux_total*y/Iy + Muy_total*x/Ix;

        if (footingType === 'concentric') {
            const Lvx = (L - cx)/2, Lvy = (B - cy)/2;
            const q_face_y = pf(cx/2, 0), q_edge_y = pf(L/2, 0);
            max_Muy = Math.max(max_Muy, B*((q_face_y - q_edge_y)*Lvx*Lvx/3 + q_edge_y*Lvx*Lvx/2));
            const q_face_x = pf(0, cy/2), q_edge_x = pf(0, B/2);
            max_Mux = Math.max(max_Mux, L*((q_face_x - q_edge_x)*Lvy*Lvy/3 + q_edge_x*Lvy*Lvy/2));
            max_Vy = Math.max(max_Vy, pf(cx/2+d, 0)*B*(Lvx-d));
            max_Vx = Math.max(max_Vx, pf(0, cy/2+d)*L*(Lvy-d));
            max_Vp = Math.max(max_Vp, c.Pu - q_avg_u*(cx+d)*(cy+d));
        } else if (footingType === 'eccentric') {
            const Lvx = eccentricEdge === 'L' ? L - cx : (L - cx)/2;
            const Lvy = eccentricEdge === 'B' ? B - cy : (B - cy)/2;
            max_Muy = Math.max(max_Muy, B*((pf(L/2,0) - pf(-L/2+cx,0))*Lvx*Lvx/3 + pf(-L/2+cx,0)*Lvx*Lvx/2));
            max_Mux = Math.max(max_Mux, L*((pf(0,B/2) - pf(0,-B/2+cy))*Lvy*Lvy/3 + pf(0,-B/2+cy)*Lvy*Lvy/2));
            max_Vy = Math.max(max_Vy, pf(cx/2+d,0)*B*(Lvx-d));
            max_Vx = Math.max(max_Vx, pf(0,cy/2+d)*L*(Lvy-d));
            max_Vp = Math.max(max_Vp, c.Pu - q_avg_u*(cx+d/2)*(cy+d));
        } else {
            const Lvx = L - cx, Lvy = B - cy;
            max_Muy = Math.max(max_Muy, B*((pf(L/2,0) - pf(-L/2+cx,0))*Lvx*Lvx/3 + pf(-L/2+cx,0)*Lvx*Lvx/2));
            max_Mux = Math.max(max_Mux, L*((pf(0,B/2) - pf(0,-B/2+cy))*Lvy*Lvy/3 + pf(0,-B/2+cy)*Lvy*Lvy/2));
            max_Vy = Math.max(max_Vy, pf(cx/2+d,0)*B*(Lvx-d));
            max_Vx = Math.max(max_Vx, pf(0,cy/2+d)*L*(Lvy-d));
            max_Vp = Math.max(max_Vp, c.Pu - q_avg_u*(cx+d/2)*(cy+d/2));
        }
    });

    const factor = metodo === 'winkler' ? (soilType === 'clay' ? 1.15 : 0.9) : metodo === 'pasternak' ? (soilType === 'clay' ? 1.1 : 0.95) : 1;
    max_Mux *= factor; max_Muy *= factor; max_Vx *= factor; max_Vy *= factor; max_Vp *= (metodo === 'winkler' ? (soilType === 'sand' ? 1.1 : 0.9) : metodo === 'pasternak' ? (soilType === 'sand' ? 1.05 : 0.95) : 1);

    const phiVcx = phi_shear * 0.17 * Math.sqrt(fc) * B * d * 1000;
    const phiVcy = phi_shear * 0.17 * Math.sqrt(fc) * L * d * 1000;
    let bo;
    if (footingType === 'concentric') bo = 2*((cx+d)+(cy+d));
    else if (footingType === 'eccentric') bo = (eccentricEdge === 'L') ? (cy+d)+2*(cx+d/2) : (cx+d)+2*(cy+d/2);
    else bo = (cx+d/2)+(cy+d/2);
    const phiVp = phi_shear * 0.33 * Math.sqrt(fc) * bo * d * 1000;

    function calculatePhiMn_max(fc, width_m, d_m) {
        const b_mm = width_m*1000, d_mm = d_m*1000;
        let beta1 = fc > 28 ? Math.max(0.65, 0.85 - 0.05*(fc-28)/7) : 0.85;
        const c_max_mm = 0.375*d_mm, a_max_mm = beta1*c_max_mm;
        const Mn_max_Nmm = 0.85*fc*a_max_mm*b_mm*(d_mm - a_max_mm/2);
        return (phi_flex*Mn_max_Nmm)/1e6;
    }
    const phiMn_max_x = calculatePhiMn_max(fc, L, d);
    const phiMn_max_y = calculatePhiMn_max(fc, B, d);

    function calculateAs(Mu, width_m, h_m, d_m) {
        const Mu_Nmm = Mu*1e6, b_mm = width_m*1000, h_mm = h_m*1000, d_mm = d_m*1000;
        if ((2*Mu_Nmm)/(phi_flex*0.85*fc*b_mm*d_mm**2) >= 1) return {as_calc: Infinity, as_req: Infinity, obs:'Aumentar sección'};
        let as_calc = (0.85*fc*b_mm*d_mm/fy)*(1 - Math.sqrt(1 - (2*Mu_Nmm)/(phi_flex*0.85*fc*b_mm*d_mm**2)));
        const as_min = 0.0018*b_mm*h_mm;
        const as_req = Math.max(as_calc, as_min);
        return { as_calc: as_calc/100, as_req: as_req/100, obs: as_req > as_calc ? 'Rige Acero Mínimo' : 'Rige por Flexión' };
    }
    const As_x = calculateAs(max_Muy, B, h, d);
    const As_y = calculateAs(max_Mux, L, h, d);

    ultimoResultado = {
        metodo, footingType, eccentricEdge, L, B, h, cx, cy,
        fc, fy, q_adm, k_s, G_s, asentamiento_adm,
        diamX, sepX, diamY, sepY,
        q_max_s, p_min_s, pressureFn,
        asentamiento_s, ex_total, ey_total, kern_x, kern_y,
        max_Mux, max_Muy, max_Vx, max_Vy, max_Vp,
        phiVcx, phiVcy, phiVp, phiMn_max_x, phiMn_max_y,
        As_x, As_y, q_avg_s, Ps, d
    };

    renderizarResultados(ultimoResultado);
    actualizarBarraEstado(true);
    document.getElementById('btnInforme').disabled = false;
    document.getElementById('btnPDF').disabled = false;
    switchTab('resultados');
}
